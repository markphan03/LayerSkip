class Solution {

class Solution:
    def twoSum(self, nums: List[int]) -> List[int]:
        if not nums:
            return []
        l = [0]
        for i in range(len(nums)):
            for j in range(i+1, len(nums)):
                if nums[i] + nums[j] == nums[i]:
                    return [i, j]
                elif nums[i] + nums[j] == nums[j]:
                    return [i, j]
                else:
                    l.append(j)
        return l
\end{code}
